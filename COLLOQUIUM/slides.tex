    \begin{columns}
        \begin{column}{0.5\textwidth}
           \visible<1-2>{
            \textbf{Measurement:}
            \begin{align*}
                \mathbf{T}\,\vec{x}=\vec{b}
            \end{align*}

            \textbf{Bolometer tomography:}
            \begin{itemize}
                \item 61 line-integrated measurements $\vec{b}$
                \item 2D emissivity $\vec{x}$: $\sim$4500 pixels
                \item \textit{Geometry} $\mathbf{T}$ has high condition
                \visible<2>{\item \color{red}{ill-posed, requires regularization}}
            \end{itemize}
            }
        \end{column}

        \begin{column}{0.5\textwidth}
            \visible<2>{
            \textbf{Problem:}
            \begin{align*}
                \underset{\eta}{\text{min}}&\left\|\left(\mathbf{T}^{\intercal}\mathbf{T}+\mu^{-1}\mathbf{H}\right)^{-1}\mathbf{T}^{\intercal}\vec{b}\,\right\|\,\rightarrow\,\vec{x}%
            \end{align*}

            \textbf{Fisher Information:}
            \begin{align*}
                I\ix{F}=\int\frac{1}{x}\left(\frac{\partial x}{\partial \vec{r}}\right)^2 d\vec{r}\,\leftrightarrow\,%
                \mathbf{H}\sim\frac{1}{x}\pmb{\Delta}^{\intercal}\pmb{\Delta}
            \end{align*}

            limiting error on reconstruction:
            \begin{equation*}
                \sigma\ix{x}\le\frac{1}{I\ix{F}}
            \end{equation*}
            }
        \end{column}
    \end{columns}

        }%

        \visible<4>{%
        \begin{itemize}
            \item Extensive artificial emissivity profile testing \& parameter variation
            \item Reconstruction quality $\ge80\%$ for well-constrained phantoms
            \item Successfull with experimental data \& power balance calculations
        \end{itemize}
        }%
    \end{enumerate}

\begin{frame}
    \frametitle{In-Situ Wheatstone Bridge Calibration}
    \backgroundlogo%

    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Ohmic Heating Method:}
            \begin{enumerate}
                \item Measure baseline current I₀
                \item Apply U$_{\text{cal}}$ = 1.2 V, 2.5 V
                \item Duration: 1.6 s per stage
                \item Measure current evolution I(t)
                \item Fit exponential decay
            \end{enumerate}

            \vspace{0.5em}
            \textbf{Extracted Parameters:}
            \begin{align*}
                R_{M} &= 2\left(\frac{U_{\text{cal}}}{I(\infty)-I_{0}}-R_{L}-R_{C}\right) \\
                \Delta I(t) &= -\Delta I(0)[1-\exp(-t/\tau_{M})] \\
                \kappa_{M} &= \frac{\Delta I^{4}(0)}{4I(\infty)(I(\infty)-I_{0})}
            \end{align*}

            \vspace{0.5em}
            \textbf{Circuit Parameters:}
            \begin{itemize}
                \item R$_L$ = 10 Ω (load resistor)
                \item R$_C$ = 40 Ω (cable resistance)
                \item C$_C$ = 2 nF (cable capacitance)
            \end{itemize}
        \end{column}

% %                 \item Measurement (M) and Reference (R) absorbers
% %                 \item AC excitation: U$_{\text{ex}}$ = 5 V (kHz range)
% %                 \item Balanced bridge: R$_M$ = R$_R$ = R $\Rightarrow$ $\Delta$U = 0
% %                 \item Incident power heats M $\Rightarrow$ R$_M$ increases
% %             \end{itemize}
% %
% %             \vspace{0.3cm}
% %             \textbf{Bridge Imbalance:}
% %             \begin{align*}
% %                 \frac{\Delta U}{U_{\text{ex}}} &= \frac{R+\Delta R}{2R+\Delta R} - \frac{R}{2R+\Delta R} \\
% %                 &\approx \frac{\Delta R}{2R}
% %             \end{align*}
% %
% %             \vspace{0.3cm}
% %             \textbf{Temperature-Resistance Relation:}
% %             \begin{itemize}
% %                 \item Linear: $\Delta R = \alpha R \Delta T$
% %                 \item Pt meanders: $\alpha$ = temperature coefficient
% %             \end{itemize}
% %         \end{column}
% %
% %         \begin{column}{0.5\textwidth}
% %             \begin{figure}
% %                 \centering
% %                 \includegraphics[width=0.9\textwidth]{../THESIS/content/figures/chapter1/wheatstone3.pdf}
% %                 \caption*{Wheatstone bridge schematic}
% %             \end{figure}
% %
% %             \vspace{0.3cm}
% %             \textbf{Power Balance:}
% %             \begin{align*}
% %                 \kappa\frac{d(\Delta T)}{dt} &= P_{\text{bol}} - \kappa\frac{\Delta T}{\tau}
% %             \end{align*}
% %
% %             \textbf{Final Bolometer Equation:}
% %             \begin{align*}
% %                 P_{\text{bol}} &= \frac{2R\kappa}{U_{\text{ex}}\alpha R\tau}\left(\Delta U + \tau\frac{d(\Delta U)}{dt}\right)
% %             \end{align*}
% %
% %             Where:
% %             \begin{itemize}
% %                 \item $\kappa$: Heat capacity
% %                 \item $\tau$: Cooling time constant
% %                 \item R: Absorber resistance
% %             \end{itemize}
% %         \end{column}
% %     \end{columns}
% %
% %     % TP: AC excitation removes 1/f noise - critical for small signal detection
% %     % TP: Bridge imbalance doubles signal-to-noise since both M absorbers respond
% %     % TP: Temperature change has two components: steady-state and transient (derivative term)
% %     % TP: Cooling time τ determines temporal response - faster cooling = better time resolution
% %     % TP: Heat capacity κ determines sensitivity - lower κ = higher temperature change for same power
% %     % TP: All parameters (R, κ, τ) must be calibrated in-situ before each campaign
