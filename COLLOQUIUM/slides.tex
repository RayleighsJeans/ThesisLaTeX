            \only<2-3>{

                \begin{equation*}
                    P_M=F_M\cdot\left(\frac{d(\Delta\widetilde{U}_M)}{dt}+f_M \Delta\widetilde{U}_M\right)
                \end{equation*}

                \begin{equation*}
                    P\ix{rad}=\frac{V\ix{P,tor}}{V_C}\sum_M \frac{P_M V_M}{K_M}

            }

    \frametitle{Minimum Fisher Regularization}
    \backgroundlogo%

    \begin{columns}
        \begin{column}{0.5\textwidth}
           \visible<1-2>{
            \textbf{Measurement:}
            \begin{align*}
                \mathbf{T}\,\vec{x}=\vec{b}
            \end{align*}

            \textbf{Bolometer tomography:}
            \begin{itemize}
                \item 61 line-integrated measurements $\vec{b}$
                \item 2D emissivity $\vec{x}$: $\sim$4500 pixels
                \item \textit{Geometry} $\mathbf{T}$ has high condition
                \visible<2>{\item \color{red}{ill-posed, requires regularization}}
            \end{itemize}
            }
        \end{column}

        \begin{column}{0.5\textwidth}
            \visible<2>{
            \textbf{Problem:}
            \begin{align*}
                \underset{\eta}{\text{min}}&\left\|\left(\mathbf{T}^{\intercal}\mathbf{T}+\mu^{-1}\mathbf{H}\right)^{-1}\mathbf{T}^{\intercal}\vec{b}\,\right\|\,\rightarrow\,\vec{x}%
            \end{align*}

            \textbf{Fisher Information:}
            \begin{align*}
                I\ix{F}=\int\frac{1}{x}\left(\frac{\partial x}{\partial \vec{r}}\right)^2 d\vec{r}\,\leftrightarrow\,%
                \mathbf{H}\sim\frac{1}{x}\pmb{\Delta}^{\intercal}\pmb{\Delta}
            \end{align*}

            limiting error on reconstruction:
            \begin{equation*}
                \sigma\ix{x}\le\frac{1}{I\ix{F}}
            \end{equation*}
            }
        \end{column}
    \end{columns}

    % TP: Tomography problem: invert line-integrated measurements (Tx=b) to reconstruct 2D emissivity distribution x
    % TP: Ill-posed inverse problem: 61 measurements but ~4500 pixels (30 radial × 150 poloidal grid) - severely underdetermined
    % TP: Geometry matrix T: encodes etendue-weighted line integrals, condition number ~10⁶ indicates strong ill-posedness
    % TP: Regularization required: Minimum Fisher Regularization (MFR) with Tikhonov-type penalty minimizes ||Tx-b||² + λ⁻¹||Hx||²
    % TP: Fisher information I_F = ∫(1/x)(∂x/∂r)² dr measures gradient content - lower I_F = smoother solution
    % TP: Regularization matrix H ∝ (1/x)∇ᵀ∇ penalizes large gradients weighted by local emissivity
    % TP: Radially dependent anisotropy (RDA): different smoothness in radial vs poloidal directions, k_ani(r) varies with radius
    % TP: Core (k_ani≈2-10): favors smooth, flux-surface aligned emission; Edge (k_ani≈0.1-0.5): allows localized, anisotropic structures
    % TP: Iterative solution: converges in N_T=10-20 iterations, regularization parameter λ chosen by L-curve method
    % TP: Computational grid: 30×150 cells (radial×poloidal), domain extends to 1.3²V_P to handle boundary radiation

\begin{frame}
    \frametitle{Plasma Transport at W7-X}
    \backgroundlogo%

    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Transport Mechanisms:}
            \begin{itemize}
                \item \textbf{Classical}: Coulomb collisions, $D\propto1/B^{2}$
                \item \textbf{Neoclassical}: Trapped particles, banana orbits, $\nabla B$ drifts
                \item \textbf{Anomalous}: Turbulence (ITG, ETG), $D_{\text{an}}\gg D_{\text{neo}}$
            \end{itemize}

            \textbf{Radial Particle Flux:}
            \begin{align*}
                \Gamma_q &= -D_q \nabla n_q + v_q n_q \\
                D_q &= D_{\text{neo},q} + D_{\text{an},q} \\
                v_q &= v_{\text{neo},q} + v_{\text{an},q}
            \end{align*}
        \end{column}

        \begin{column}{0.5\textwidth}
            \textbf{Neoclassical Regimes:}
            \begin{itemize}
                \item \textbf{Banana} (low $\nu$): Trapped particles dominant, W7-X optimized
                \item \textbf{Electron-root} ($T_e\gg T_i$): Reduced transport $\propto\sqrt{\nu}$
                \item \textbf{Ion-root} (higher $\nu$): Transport $\propto$ 1/$\nu$
                \item \textbf{Pfirsch-Schlüter} (high $\nu$): $D_{\text{PS}}\propto D_{\text{classical}}$
            \end{itemize}

            \textbf{W7-X Optimization:}
            \begin{itemize}
                \item Reduced neoclassical transport in low collisionality
                \item Still 1-2 orders higher than tokamaks
            \end{itemize}
        \end{column}
    \end{columns}

    % TP: Neoclassical transport regimes: (1) Banana (low ν, trapped particles), (2) Electron-root (T_e>>T_i, reduced transport ∝√ν), (3) Ion-root (higher ν, ∝1/ν), (4) Pfirsch-Schlüter (high ν, ∝D_classical)

\begin{frame}
    \frametitle{Bolometer Equation Derivation}
    \backgroundlogo

    \begin{columns}
        \begin{column}{0.5\textwidth}
            \only<1>{
                \textbf{Wheatstone Bridge Concept:}
                \begin{itemize}
                    \item Measurement (M) and Reference (R)
                    \item AC excitation: U$_{\text{ex}}$ = 5 V (kHz range)
                    \item Incident power heats M $\Rightarrow$ R$_M$ increases
                    \item Linear: $\Delta R = \alpha R \Delta T$
                \end{itemize}

                \vspace{0.3cm}
                \textbf{Bridge Imbalance:}
                \begin{align*}
                    \frac{\Delta U}{U_{\text{ex}}} &= \frac{R+\Delta R}{2R+\Delta R} - \frac{R}{2R+\Delta R} \approx \frac{\Delta R}{2R}\\
                    \Delta T &= \frac{\Delta R}{\alpha R \Delta T}=\frac{2\Delta U}{\alpha U\ix{ex}}
                \end{align*}
            }

            \only<2>{
                \textbf{Power Balance:}
                \begin{align*}
                    \kappa\frac{d(\Delta T)}{dt} &= P_{\text{bol}} - \kappa\frac{\Delta T}{\tau}
                \end{align*}

                \textbf{Final Bolometer Equation:}
                \begin{align*}
                    P_{\text{bol}} &= \frac{2R\kappa}{U_{\text{ex}}\alpha R\tau}\left(\Delta U + \tau\frac{d(\Delta U)}{dt}\right)
                \end{align*}

                Where:
                \begin{itemize}
                    \item $\kappa$: Heat capacity
                    \item $\tau$: Cooling time constant
                    \item R: Absorber resistance
                \end{itemize}
            }
        \end{column}

        \begin{column}{0.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.9\textwidth]{../THESIS/content/figures/chapter1/wheatstone3.pdf}
                \gcaption{Wheatstone bridge schematic}
            \end{figure}
        \end{column}
    \end{columns}

    % TP: AC excitation (U_ex=5V, kHz range): removes 1/f noise, critical for μV-level signal detection in fusion environment
    % TP: Bridge configuration: two measurement absorbers (M) vs two reference resistors (R), imbalance ΔU ∝ ΔR_M
    % TP: Linear approximation: ΔU/U_ex ≈ ΔR/(2R) valid for ΔR<<R, typical ΔR/R~10⁻⁴ during plasma
    % TP: Temperature coefficient: α=3.4×10⁻³ K⁻¹ for gold, relates resistance change to temperature: ΔR=αRΔT
    % TP: Power balance equation: κ(dΔT/dt) = P_bol - κΔT/τ, includes heating (P_bol) and cooling (τ⁻¹) terms
    % TP: Steady-state term: ΔU represents equilibrium temperature, proportional to absorbed power
    % TP: Transient term: τ(dΔU/dt) captures rapid power changes, essential for temporal resolution <100ms
    % TP: Cooling time τ=110.57ms: determines frequency response, faster cooling → better time resolution but lower sensitivity
    % TP: Heat capacity κ=0.724A²: determines sensitivity, lower κ → larger ΔT for same power but slower response
    % TP: Trade-off: τ and κ inversely related, optimized for fusion timescales (10ms-10s) and power levels (0.1-10 MW/m²)
    % TP: Calibration necessity: R, κ, τ vary ±5-10% between detectors, in-situ calibration mandatory before each campaign
\end{frame}

\begin{frame}
    \frametitle{STRAHL Basics}
    \backgroundlogo

    \only<1>{
        \begin{flushleft}
            \textbf{A Priori Inputs Required:}
        \end{flushleft}
        \begin{itemize}
            \item[1.] Background plasma profiles: $n_e(r,t)$ electron density, $T_e(r,t)$ electron temperature from Thomson scattering
            \item[2.] Magnetic geometry: flux surfaces, connection lengths, effective radii (from equilibrium reconstruction)
            \item[3.] Transport coefficients: $D(r)$ diffusion, $v(r)$ convection (either fitted to data or assumed from theory)
            \item[4.] Impurity source: location (divertor, wall, gas valve), rate $\Gamma_s(t)$, species (C, O, N, Ne, etc.)
            \item[5.] Atomic data: ionization rates $Q_{ion}(T_e)$, recombination rates $Q_{rec}(T_e,n_e)$, charge exchange $Q_{cex}$ from ADAS database
            \item[6.] Neutral density: $n_H(r)$ hydrogen/deuterium profile for charge exchange reactions
        \end{itemize}
    }
    \only<2>{
        \begin{flushleft}
            \textbf{Physics Captured:}
        \end{flushleft}
        \begin{itemize}
            \item Radial transport: diffusion (turbulent) and convection (drift, neoclassical) move impurities inward/outward
            \item Ionization/recombination: balance determines charge state distribution $Z(r)$, depends on local $T_e$
            \item Charge exchange: neutrals capture electrons from ions, affects edge ionization balance
            \item Radiation: each charge state emits line radiation + continuum, summed to get total $P_{rad}(r)$
            \item Flux-surface averaging: D* and v* are poloidally averaged, accounts for stellarator 3D geometry approximately
        \end{itemize}
    }

   \only<3>{
        \begin{flushleft}
            \textbf{Key Assumptions:}
        \end{flushleft}
        \begin{itemize}
            \item Trace impurity: $n_Z\ll n_e$, impurities don't affect background plasma (valid for $Z_{eff}<2$)
            \item Coronal equilibrium: ionization balance faster than transport (valid for core, breaks down in edge)
            \item No neutral transport: assumes neutrals instantly ionized or lost (simplified, neglects recycling details)
            \item Steady-state background: $n_e(r)$, $T_e(r)$ evolve slowly compared to impurity transport timescales
        \end{itemize}
    }

   \only<4>{
        \begin{flushleft}
            \textbf{Outputs:}
        \end{flushleft}
        \begin{itemize}
            \item Impurity density profiles: $n_i,\,\,Z(r,t)$ for each charge state $Z$
            \item Radiation profiles: $P_{rad}(r,t) = \sum_Z n_{i,Z(r,t)}\times L_Z(T_e)$ where $L_Z$ is cooling rate
            \item Fractional abundance: $f_Z(r) = n_i,Z / \sum_Z n_{i,Z}$, shows dominant charge states
            \item Integrated quantities: total radiated power, $Z_{eff}$ contribution, impurity confinement time
        \end{itemize}
    }

   \only<5>{
        \begin{flushleft}
            \textbf{Validation Strategy:}
        \end{flushleft}
        \begin{itemize}
            \item Compare STRAHL $P_{rad}(r)$ with bolometry measurements (tomographic reconstruction)
            \item Adjust $D(r)$ and $v(r)$ to match experimental radiation profiles
            \item Check charge state ratios against spectroscopy (e.g., C$^{2+}$/$C^{4+}$ from visible lines)
            \item Verify temporal evolution during gas puff experiments
        \end{itemize}
   }

        v^{\ast}=&\left\langle v\left(\theta\right)\vert\nabla r\vert\right\rangle\ix{FS}=\frac{1}{4\pi^{2}R\ix{M}r}\displaystyle\oint\ix{FS}v\diff S\,\,.\\
        \begin{split}
            S_{\text{i,}Z}=-&n_{\text{i,}Z}\left(n\ix{e}Q\head{ion}_{\text{i,}Z}+n\ix{e}Q_{\text{i,}Z}\head{rec}+n\ix{H}Q_{\text{i,}Z}\head{cex}\right)+n\ix{e}Q_{\text{i,}Z-1}\head{ion}n_{\text{i,}Z-1}+\\%

        \end{split}
    \end{align*}

        \end{split}%

        \end{split}

            \textbf{FFT Correlation:}
            \begin{itemize}
                \item Uses spectral correlation between $P\ix{rad}$ and $P\ix{pred}$ with DFFT
                \item Cross-spectral density, coherence-based metric
            \end{itemize}

            \textbf{Mathematical Formulation:}
            \begin{align*}
                \gamma_i &= \mathcal{F}(P\ix{rad})(\omega)\qquad%
                \gamma_j  = \mathcal{F}(P\ix{pred})(\omega) \\
                g\ix{i,j}\left(\omega\right)=\frac{1}{T}&\int_{-\infty}^{\infty}\gamma\ix{i}\left(\omega\right)\gamma\ix{j}\left(\omega-\nu\right)\diff\nu\\
                \varphi(\omega) &= \sqrt{|g\ix{i,j}^2|/|\gamma_i\gamma_j|} \Rightarrow \vartheta = \int \varphi(\omega) d\omega
            \end{align*}

    % TP: FFT correlation: spectral analysis using Discrete Fourier Transform (DFFT) of P_rad and P_pred time series
    % TP: Mathematical formulation: γ_i = F(P_rad)(ω), γ_j = F(P_pred)(ω), transforms to frequency domain
    % TP: Cross-spectral density: g_i,j(ω) = (1/T)∫γ_i(ω)γ_j(ω-ν)dν, measures frequency-dependent correlation
    % TP: Coherence metric: φ(ω) = √[|g_i,j²|/|γ_iγ_j|], normalized spectral correlation (0-1 range)
    % TP: Integrated deviation: ϑ = ∫φ(ω)dω, total spectral correlation across all frequencies
    % TP: Frequency-domain advantages: identifies oscillatory behavior, separates fast/slow dynamics, filters noise

    % TP: Global maximum: at upper inboard X-point viewing channels, highest spectral correlation with P_rad
    % TP: Spectral focus: excludes absolute value contributions, emphasizes relative frequency content
    % TP: Identifies oscillations: gas puff cycles (~1-5 Hz), MHD activity (~10-100 Hz), drift waves (~kHz)
    % TP: Complements time-domain: FFT reveals frequency-specific sensitivities not visible in time-averaged metrics
    % TP: Feedback implications: spectral analysis guides controller design, identifies problematic frequency ranges
