                \begin{itemize}
                    \item Power balance calculations
                    \item \color{ipp}{\textbf{Tomographic reconstruction}}
                    \item \color{ipp}{\textbf{Real-time feedback control}}
                \end{itemize}
                \color{red}{\Rightarrow\,\,controlled radiative cooling in the plasma edge}
            }

        \begin{column}{0.45\textwidth}

            }

            \only<2-3>{
                \textbf{Power Per Detector:}
                \begin{equation*}
                    P_M=F_M\cdot\left(\frac{d(\Delta\widetilde{U}_M)}{dt}+f_M \Delta\widetilde{U}_M\right)

                \begin{equation*}

\section{Tomography}

\begin{frame}
    \frametitle{Minimum Fisher Regularization}
    \backgroundlogo%

    \begin{columns}
        \begin{column}{0.5\textwidth}
           \only<1-2>{
            \textbf{Measurement:}
            \begin{align*}
                \mathbf{T}\,\vec{x}=\vec{b}
            \end{align*}

            \textbf{Bolometer tomography:}
            \begin{itemize}
                \item 61 line-integrated measurements $\vec{b}$
                \item 2D emissivity $\vec{x}$: $\sim$4500 pixels
                \item \textit{Geometry} $\mathbf{T}$ has high condition
                \item \color{red}{ill-posed, requires regularization}
            \end{itemize}
            }

            \only<3-4>{
            \begin{block}{MFR + RDA Advantages}
                \begin{itemize}
                    \item Robust to noisy data
                    \item Incorporates a priori knowledge
                    \item Stable convergence
                \end{itemize}
            \end{block}
           }

           \only<4>{
            \centering
            \includegraphics[width=\textwidth]{%
                ../THESIS/content/figures/chapter4/kAni_profile_comparison.pdf}
            }
        \end{column}

        \begin{column}{0.45\textwidth}
            \only<2>{
            \textbf{Problem:}
            \begin{align*}
                \underset{\eta}{\text{min}}&\left\|\left(\mathbf{T}^{\intercal}\mathbf{T}+\mu^{-1}\mathbf{H}\right)^{-1}\mathbf{T}^{\intercal}\vec{b}\,\right\|\,\rightarrow\,\vec{x}%
            \end{align*}

            \textbf{Fisher Information:}
            \begin{align*}
                I\ix{F}=\int\frac{1}{x}\left(\frac{\partial x}{\partial \vec{r}}\right)^2 d\vec{r}\,\leftrightarrow\,%
                \mathbf{H}\sim\frac{1}{x}\pmb{\Delta}^{\intercal}\pmb{\Delta}
            \end{align*}

            limiting error on reconstruction:
            \begin{equation*}
                \sigma\ix{x}\le\frac{1}{I\ix{F}}
            \end{equation*}
            }

            \only<3-4>{%
            \textbf{Radially Dependent Anisotropy:}
            \begin{itemize}
                \item Tailored weighting to poloidal gradient
                \item $k\ix{ani}<1$: favours anisotropy
                \item $k\ix{ani}>1$: favours smooth emissivity
            \end{itemize}

            \vspace{-1em}
            \begin{align*}
                \mathbf{H}&\to\mathbf{H}\ix{ani}\\%
                \pmb{\Delta}&=\pmb{\nabla}\ix{r}^{\intercal}(\cdot)\pmb{\nabla}\ix{r}+\pmb{\nabla}_{\vartheta}^{\intercal}(\cdot)\pmb{\nabla}_{\vartheta}\\%
                \pmb{\nabla}_{\vartheta}&\to\pmb{\widetilde{\nabla}}_{\vartheta}=\frac{1}{k\ix{ani}\Delta r}\mathbf{D}\ix{dia}^{(0)}\pmb{\nabla}_{\vartheta}
            \end{align*}
            }
        \end{column}
    \end{columns}

    % TP: Tomography inverts line-integrated measurements to get 2D radiation distribution
    % TP: This is ill-posed - 80 measurements but ~4500 pixels to reconstruct, needs regularization
    % TP: MFR finds the smoothest solution consistent with measurements using Fisher information theory
    % TP: Fisher information I_F measures how much gradient information is in the solution
    % TP: Radially dependent anisotropy allows different smoothness in core vs edge - captures physics better
    % TP: Core (α≈10): smooth, flux-surface aligned; Edge (α≈0.1): localized, allows sharp gradients
    % TP: Iterative solution converges in ~10-20 iterations with regularization parameter λ
    % TP: Grid: 30×150 pixels, domain: 1.3² times plasma volume for boundary handling
\end{frame}
